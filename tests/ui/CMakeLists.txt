cmake_minimum_required(VERSION 3.20)
project(mybot_ui_host C CXX)

set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory(../../managed_components/lvgl__lvgl lvgl)

add_executable(ui_preview test_ui.c ../../main/pair_code_font.c)
add_executable(power_tests test_power.c)
foreach(host_target ui_preview power_tests)
    target_include_directories(${host_target} PRIVATE stubs ../host/stubs ../../main)
    target_link_libraries(${host_target} PRIVATE lvgl m)
    target_compile_features(${host_target} PRIVATE c_std_11)
    target_compile_options(${host_target} PRIVATE -Wall -Wextra -Werror)
    target_link_options(${host_target} PRIVATE -fsanitize=address,undefined)
endforeach()
foreach(ui_target lvgl ui_preview power_tests)
    target_compile_options(${ui_target} PRIVATE -g -fsanitize=address,undefined -fno-omit-frame-pointer)
endforeach()
# LVGL's generic mask callback casts its concrete radius descriptor to void *.
if(CMAKE_C_COMPILER_ID MATCHES "Clang")
    target_compile_options(lvgl PRIVATE -fno-sanitize=function)
endif()
